# Salah Time (সালাহ টাইম)

Flutter user app. Package: `com.bjdfarhma.salah`, Firebase project: `salah-time-d5662`.

## Local build
    flutter create --platforms=android --org com.bjdfarhma --project-name salah .
    python3 tool/patch_android.py
    flutter pub get
    dart run flutter_launcher_icons
    flutter build apk --release

## Firebase setup you must do (Console)
1. Realtime Database -> Rules: allow public read on `posters` (poster slider):
   `{ "rules": { "posters": { ".read": true, ".write": false } } }`
   Poster format: `posters/0 = { "title": "...", "subtitle": "...", "image": "https://..." }`
   Dua and quiz can also be managed from the admin panel: `duas` and `quiz` (public read, admin-only write).
   If `duas` / `quiz` are empty the app uses its built-in defaults.
2. Cloud Messaging: users subscribe to topic `all`. Send test message to topic `all` from Console.
3. Add your release SHA-1 later only if you add Google sign-in (not used now).

## Notes
- Quran text/translation and audio load from api.alquran.cloud / islamic.network; each surah is cached after first open (offline later).
- Hijri date may differ 1 day from Bangladesh sighting: adjust in Settings.
- Home-screen widget needs native Android code (planned next step).

  static final AppState i = AppState._();
  AppState._();

  late SharedPreferences prefs;
  String lang = 'bn';
  ThemeMode mode = ThemeMode.system;
  double fontScale = 1.0;
  bool azan = true;
  bool onboarded = false;
  int hijriAdj = 0;
  double lat = 23.8103, lng = 90.4125;
  String place = 'Dhaka';
  int tasbihCount = 0, tasbihTarget = 33, tasbihTotal = 0, quizBest = 0;
  Map<String, bool> prayerOn = {for (final k in prayerKeys) k: true};

  Future<void> load() async {
    prefs = await SharedPreferences.getInstance();
    lang = prefs.getString('lang') ?? 'bn';
    mode = ThemeMode.values[prefs.getInt('mode') ?? 0];
    fontScale = prefs.getDouble('font') ?? 1.0;
    azan = prefs.getBool('azan') ?? true;
    onboarded = prefs.getBool('onboarded') ?? false;
    hijriAdj = prefs.getInt('hijriAdj') ?? 0;
    lat = prefs.getDouble('lat') ?? lat;
    lng = prefs.getDouble('lng') ?? lng;
    place = prefs.getString('place') ?? place;
    tasbihCount = prefs.getInt('tc') ?? 0;
    tasbihTarget = prefs.getInt('tt') ?? 33;
    tasbihTotal = prefs.getInt('ttl') ?? 0;
    quizBest = prefs.getInt('qb') ?? 0;
    for (final k in prayerKeys) {
      prayerOn[k] = prefs.getBool('p_$k') ?? true;
    }
  }

  void setLang(String v) { lang = v; prefs.setString('lang', v); notifyListeners(); }
  void setMode(ThemeMode m) { mode = m; prefs.setInt('mode', m.index); notifyListeners(); }
  void setFont(double v) { fontScale = v; prefs.setDouble('font', v); notifyListeners(); }
  void setAzan(bool v) { azan = v; prefs.setBool('azan', v); notifyListeners(); }
  void setHijriAdj(int v) { hijriAdj = v; prefs.setInt('hijriAdj', v); notifyListeners(); }
  void setOnboarded() { onboarded = true; prefs.setBool('onboarded', true); }
  void setPrayerOn(String k, bool v) { prayerOn[k] = v; prefs.setBool('p_$k', v); notifyListeners(); }
  void setLocation(double a, double b, String p) {
    lat = a; lng = b; place = p;
    prefs.setDouble('lat', a); prefs.setDouble('lng', b); prefs.setString('place', p);
    notifyListeners();
  }
  void saveTasbih() {
    prefs.setInt('tc', tasbihCount); prefs.setInt('tt', tasbihTarget); prefs.setInt('ttl', tasbihTotal);
    notifyListeners();
  }
  void setQuizBest(int v) { quizBest = v; prefs.setInt('qb', v); }

  // Notification inbox
  List<Map<String, dynamic>> inbox() {
    final raw = prefs.getStringList('inbox') ?? [];
    return raw.map((e) => jsonDecode(e) as Map<String, dynamic>).toList();
  }
  void addInbox(String title, String body) {
    final l = prefs.getStringList('inbox') ?? [];
    l.insert(0, jsonEncode({'t': title, 'b': body, 'd': DateTime.now().toIso8601String()}));
    prefs.setStringList('inbox', l.take(50).toList());
    notifyListeners();
  }

/// Converts digits to Bangla when the language is Bangla.
String nd(Object v) {
  final s = v.toString();
  if (AppState.i.lang != 'bn') return s;
  const bn = ['০', '১', '২', '৩', '৪', '৫', '৬', '৭', '৮', '৯'];
  return s.replaceAllMapped(RegExp(r'[0-9]'), (m) => bn[int.parse(m[0]!)]);
}

class PrayerService {
  /// All six times (with sunrise) for the given day.
  static List<PrayerTime> forDay(DateTime day) {
    final c = Coordinates(st.lat, st.lng);
    final p = CalculationMethod.karachi.getParameters();
    p.madhab = Madhab.hanafi;
    final t = PrayerTimes(c, DateComponents.from(day), p);
    return [
      PrayerTime('fajr', t.fajr.toLocal()),
      PrayerTime('sunrise', t.sunrise.toLocal()),
      PrayerTime('dhuhr', t.dhuhr.toLocal()),
      PrayerTime('asr', t.asr.toLocal()),
      PrayerTime('maghrib', t.maghrib.toLocal()),
      PrayerTime('isha', t.isha.toLocal()),
    ];
  }

  /// Next of the five obligatory prayers (skips sunrise).
  static PrayerTime next(DateTime now) {
    for (final p in forDay(now)) {
      if (p.key != 'sunrise' && p.time.isAfter(now)) return p;
    }
    return forDay(now.add(const Duration(days: 1))).first;
  }

  static PrayerTime? current(DateTime now) {
    PrayerTime? cur;
    for (final p in forDay(now)) {
      if (p.key != 'sunrise' && !p.time.isAfter(now)) cur = p;
    }
    return cur;
  }

  static Future<bool> updateLocation() async {
    try {
      if (!await Geolocator.isLocationServiceEnabled()) return false;
      var perm = await Geolocator.checkPermission();
      if (perm == LocationPermission.denied) perm = await Geolocator.requestPermission();
      if (perm == LocationPermission.denied || perm == LocationPermission.deniedForever) return false;
      final pos = await Geolocator.getCurrentPosition(
          desiredAccuracy: LocationAccuracy.low, timeLimit: const Duration(seconds: 15));
      st.setLocation(pos.latitude, pos.longitude,
          '${pos.latitude.toStringAsFixed(2)}, ${pos.longitude.toStringAsFixed(2)}');
      await Notifier.schedule();
      return true;
    } catch (_) {
      return false;
    }
  }
}

String fmtTime(DateTime t) {
  final s = DateFormat('h:mm').format(t);
  return '${nd(s)} ${t.hour < 12 ? tr('am') : tr('pm')}';
}

String fmtDur(Duration d) {
  if (d.isNegative) d = Duration.zero;
  String two(int n) => n.toString().padLeft(2, '0');
  return nd('${two(d.inHours)}:${two(d.inMinutes % 60)}:${two(d.inSeconds % 60)}');
}

/// Bangladesh revised Bangla calendar (new year on 14 April).
BnDate bnDate(DateTime dt) {
  final date = DateTime.utc(dt.year, dt.month, dt.day);
  final gy = date.isBefore(DateTime.utc(dt.year, 4, 14)) ? dt.year - 1 : dt.year;
  var days = date.difference(DateTime.utc(gy, 4, 14)).inDays;
  final ny = gy + 1;
  final leap = (ny % 4 == 0 && ny % 100 != 0) || ny % 400 == 0;
  final lens = [31, 31, 31, 31, 31, 30, 30, 30, 30, 30, leap ? 30 : 29, 30];
  var m = 0;
  while (m < 11 && days >= lens[m]) {
    days -= lens[m];
    m++;
  }
  return BnDate(gy - 593, m, days + 1);
}

class Notifier {
  static final fln = FlutterLocalNotificationsPlugin();
  static const _details = NotificationDetails(
      android: AndroidNotificationDetails('azan', 'Azan',
          channelDescription: 'Prayer time alerts', importance: Importance.max, priority: Priority.high));

  static Future<void> init() async {
    tzdata.initializeTimeZones();
    await fln.initialize(const InitializationSettings(android: AndroidInitializationSettings('@mipmap/ic_launcher')));
  }

  static Future<void> ask() async {
    await fln
        .resolvePlatformSpecificImplementation<AndroidFlutterLocalNotificationsPlugin>()
        ?.requestNotificationsPermission();
  }

  static Future<void> show(String title, String body) =>
      fln.show(DateTime.now().millisecondsSinceEpoch ~/ 1000, title, body, _details);

  static Future<void> schedule() async {
    try {
      await fln.cancelAll();
      if (!st.azan) return;
      final now = DateTime.now();
      for (var d = 0; d < 3; d++) {
        final day = now.add(Duration(days: d));
        for (final p in PrayerService.forDay(day)) {
          if (p.key == 'sunrise' || st.prayerOn[p.key] != true || !p.time.isAfter(now)) continue;
          final id = d * 10 + prayerKeys.indexOf(p.key);
          final when = tz.TZDateTime.from(p.time, tz.UTC);
          final title = '${tr(p.key)} - ${tr('app')}';
          final body = fmtTime(p.time);
          try {
            await fln.zonedSchedule(id, title, body, when, _details,
                androidScheduleMode: AndroidScheduleMode.exactAllowWhileIdle,
                uiLocalNotificationDateInterpretation: UILocalNotificationDateInterpretation.absoluteTime);
          } catch (_) {
            await fln.zonedSchedule(id, title, body, when, _details,
                androidScheduleMode: AndroidScheduleMode.inexactAllowWhileIdle,
                uiLocalNotificationDateInterpretation: UILocalNotificationDateInterpretation.absoluteTime);
          }
        }
      }
    } catch (_) {}
  }
}

class Push {

  }
}

class Remote {
  static List<Map<String, dynamic>> _parseList(String? raw) {
    if (raw == null) return [];
    try {
      final j = jsonDecode(raw);
      final list = j is List ? j : (j is Map ? j.values.toList() : <dynamic>[]);
      return list.whereType<Map>().map((e) => Map<String, dynamic>.from(e)).toList();
    } catch (_) {
      return [];
    }
  }

  /// Instant read of a list from the local cache (no network).
  static List<Map<String, dynamic>> cachedList(String path) => _parseList(st.prefs.getString(path));

  /// Fetch /<path> from Realtime Database (public read), cache it for offline, return the list.
  /// If the admin deleted everything (null) the cache is cleared too.
  static Future<List<Map<String, dynamic>>> list(String path) async {
    try {
      final r = await http.get(Uri.parse('$dbUrl/$path.json')).timeout(const Duration(seconds: 8));
      if (r.statusCode == 200) {
        final body = utf8.decode(r.bodyBytes);
        if (body.trim() == 'null') {
          await st.prefs.remove(path);
        } else {
          await st.prefs.setString(path, body);
        }
      }
    } catch (_) {}
    return cachedList(path);
  }

  /// Poster slider items from Realtime Database path /posters (cached for offline).
  static Future<List<Map<String, dynamic>>> posters() => list('posters');

  /// Fetch JSON with local cache so it keeps working offline.
  static Future<dynamic> cached(String url, String key) async {
    try {
      final r = await http.get(Uri.parse(url)).timeout(const Duration(seconds: 20));
      if (r.statusCode == 200) {
        await st.prefs.setString(key, utf8.decode(r.bodyBytes));
      }
    } catch (_) {}
    final raw = st.prefs.getString(key);
    return raw == null ? null : jsonDecode(raw);
  }
}

class _DuaScreenState extends State<DuaScreen> {
  String q = '';
  List<Dua> items = duas;

  /// Admin-managed list (Firebase /duas). Empty or missing -> built-in defaults.
  static List<Dua> _parse(List<Map<String, dynamic>> raw) {
    final l = raw.map(Dua.fromMap).where((d) => d.ar.isNotEmpty || d.bn.isNotEmpty || d.en.isNotEmpty).toList();
    return l.isEmpty ? duas : l;
  }

  @override
  void initState() {
    super.initState();
    items = _parse(Remote.cachedList('duas'));
    Remote.list('duas').then((r) {
      if (mounted) setState(() => items = _parse(r));
    });
  }

  @override
  Widget build(BuildContext context) {
    final bn = st.lang == 'bn';
    final list = items.where((d) {
      final s = q.trim().toLowerCase();
      return s.isEmpty || d.title.contains(s) || d.titleEn.toLowerCase().contains(s) || d.bn.contains(s) || d.en.toLowerCase().contains(s) || d.ar.contains(s);
    }).toList();
    return Scaffold(
      appBar: AppBar(title: Text(tr('dua'))),
      body: Column(children: [
        Padding(
          padding: const EdgeInsets.fromLTRB(16, 8, 16, 8),
          child: TextField(
            onChanged: (v) => setState(() => q = v),
            decoration: InputDecoration(
              prefixIcon: const Icon(Icons.search),
              hintText: tr('search'),
              filled: true,
              border: OutlineInputBorder(borderRadius: BorderRadius.circular(16), borderSide: BorderSide.none),
            ),
          ),
        ),
        Expanded(
          child: ListView(padding: const EdgeInsets.fromLTRB(16, 4, 16, 24), children: [

                  Text(bn ? (d.bn.isEmpty ? d.en : d.bn) : (d.en.isEmpty ? d.bn : d.en)),
                ]),
              ),
          ]),
        ),
      ]),
    );
  }
}

  @override
  void initState() {
    super.initState();
    WidgetsBinding.instance.addObserver(this);
    _load();
    // pick up admin-panel changes without restarting the app
    _poll = Timer.periodic(const Duration(seconds: 30), (_) => _load());
  }

  @override
  void dispose() {
    _poll?.cancel();
    WidgetsBinding.instance.removeObserver(this);
    super.dispose();
  }

    if (state == AppLifecycleState.resumed) _load();

      ? widget.items

              child: Column(mainAxisSize: MainAxisSize.min, crossAxisAlignment: CrossAxisAlignment.start, children: [
                Text('${m['title'] ?? ''}', style: const TextStyle(color: AppTheme.gold, fontSize: 22, fontWeight: FontWeight.w800)),

class Q {
  final String bn, en;
  final List<String> obn, oen;
  final int a;
  const Q(this.bn, this.en, this.obn, this.oen, this.a);

  /// Build a question from an admin-panel map; returns null if it is incomplete.
  static Q? tryParse(Map<String, dynamic> m) {
    List<String> strs(dynamic v) {
      final l = v is List ? v : (v is Map ? v.values.toList() : <dynamic>[]);
      return l.map((e) => '$e').toList();
    }

    final obn = strs(m['obn']), oen = strs(m['oen']);
    final bn = '${m['bn'] ?? ''}', en = '${m['en'] ?? ''}';
    final a = (num.tryParse('${m['a']}') ?? -1).toInt();
    final ob = obn.isNotEmpty ? obn : oen;
    final oe = oen.isNotEmpty ? oen : obn;
    final qb = bn.isNotEmpty ? bn : en;
    final qe = en.isNotEmpty ? en : bn;
    if (qb.isEmpty || ob.length < 2 || ob.length != oe.length || a < 0 || a >= ob.length) return null;
    return Q(qb, qe, ob, oe, a);
  }
}

class _QuizScreenState extends State<QuizScreen> {
  late List<Q> list;
  List<Q> pool = questions;
  int i = 0, score = 0;
  int? picked;
  bool done = false;

  /// Admin-managed questions (Firebase /quiz). Empty or missing -> built-in defaults.
  static List<Q> _parse(List<Map<String, dynamic>> raw) {
    final l = <Q>[];
    for (final m in raw) {
      final q = Q.tryParse(m);
      if (q != null) l.add(q);
    }
    return l.isEmpty ? questions : l;
  }

  @override
  void initState() {
    super.initState();
    pool = _parse(Remote.cachedList('quiz'));
    _start();
    Remote.list('quiz').then((r) {
      if (!mounted) return;
      setState(() {
        pool = _parse(r);
        // not started answering yet -> use the fresh questions right away
        if (i == 0 && picked == null && !done) _start();
      });
    });
  }

  void _start() {
    list = [...pool]..shuffle();
    i = 0; score = 0; picked = null; done = false;
  }

  @override
  Widget build(BuildContext context) {
    final bn = st.lang == 'bn';
    return Scaffold(
      appBar: AppBar(title: Text(tr('quiz'))),
      body: Padding(
        padding: const EdgeInsets.all(16),
        child: done ? _result() : _question(bn),
      ),
    );
  }

  Widget _result() => Center(
        child: Column(mainAxisSize: MainAxisSize.min, children: [
          Icon(Icons.emoji_events_outlined, size: 80, color: accent(context)),
          const SizedBox(height: 12),
          Text('${tr('score')}: ${nd(score)} / ${nd(list.length)}', style: const TextStyle(fontSize: 26, fontWeight: FontWeight.w800)),
          Text('${tr('best')}: ${nd(st.quizBest)}'),
          const SizedBox(height: 20),
          FilledButton(onPressed: () => setState(_start), child: Text(tr('again'))),
        ]),
      );

  Widget _question(bool bn) {
    final q = list[i];
    final opts = bn ? q.obn : q.oen;
    return ListView(children: [
      LinearProgressIndicator(value: (i + 1) / list.length, minHeight: 6, borderRadius: BorderRadius.circular(3)),
      const SizedBox(height: 18),
      Text('${nd(i + 1)} / ${nd(list.length)}', style: TextStyle(color: accent(context), fontWeight: FontWeight.w700)),
      const SizedBox(height: 8),
      Text(bn ? q.bn : q.en, style: const TextStyle(fontSize: 21, fontWeight: FontWeight.w800)),
      const SizedBox(height: 16),
      for (var k = 0; k < opts.length; k++)
        Panel(
          onTap: picked != null
              ? null
              : () => setState(() {
                    picked = k;
                    if (k == q.a) score++;
                  }),
          child: Row(children: [
            Expanded(child: Text(opts[k], style: const TextStyle(fontSize: 16, fontWeight: FontWeight.w600))),
            if (picked != null && k == q.a) const Icon(Icons.check_circle, color: Colors.green),
            if (picked == k && k != q.a) const Icon(Icons.cancel, color: Colors.red),
          ]),
        ),
      if (picked != null)
        FilledButton(
          onPressed: () => setState(() {
            if (i + 1 >= list.length) {
              done = true;
              if (score > st.quizBest) st.setQuizBest(score);
            } else {
              i++;
              picked = null;
            }
          }),
          child: Text(tr('nextQ')),
        ),
    ]);
  }

  @override
  State<SurahScreen> createState() => _SurahScreenState();

class _SurahScreenState extends State<SurahScreen> {
  List<dynamic>? ar, tx;
  bool loading = true, playing = false;
  final player = AudioPlayer();

  @override
  void initState() {
    super.initState();
    player.onPlayerComplete.listen((_) => setState(() => playing = false));
    _load();
  }

  @override
  void dispose() {
    player.dispose();
    super.dispose();
  }

  Future<void> _load() async {
    final ed = st.lang == 'bn' ? 'bn.bengali' : 'en.sahih';
    final j = await Remote.cached(
        'https://api.alquran.cloud/v1/surah/${widget.number}/editions/quran-uthmani,$ed', 'q_${widget.number}_$ed');
    if (j != null) {
      ar = j['data'][0]['ayahs'];
      tx = j['data'][1]['ayahs'];
    }
    if (mounted) setState(() => loading = false);
  }

  Future<void> _toggle() async {
    if (playing) {
      await player.stop();
      setState(() => playing = false);
    } else {
      setState(() => playing = true);
      try {
        await player.play(UrlSource('https://cdn.islamic.network/quran/audio-surah/128/ar.alafasy/${widget.number}.mp3'));
      } catch (_) {
        setState(() => playing = false);
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text(widget.name), actions: [
        IconButton(tooltip: playing ? tr('stop') : tr('play'), onPressed: _toggle, icon: Icon(playing ? Icons.stop_circle_outlined : Icons.play_circle_outline)),
      ]),
      body: loading
          ? const Center(child: CircularProgressIndicator())
          : ar == null
              ? Center(child: Padding(padding: const EdgeInsets.all(24), child: Text(tr('needNet'), textAlign: TextAlign.center)))
              : ListView.builder(
                  padding: const EdgeInsets.fromLTRB(16, 8, 16, 24),
                  itemCount: ar!.length,
                  itemBuilder: (_, i) => Panel(
                    child: Column(crossAxisAlignment: CrossAxisAlignment.start, children: [
                      Align(
                          alignment: Alignment.centerRight,
                          child: Text('${ar![i]['text']}', textDirection: TextDirection.rtl, style: const TextStyle(fontSize: 26, height: 2))),
                      const SizedBox(height: 8),
                      Text('${nd(i + 1)}. ${tx![i]['text']}'),
                    ]),
                  ),
                ),
    );
  }
